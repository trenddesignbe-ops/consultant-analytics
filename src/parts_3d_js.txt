/* ---------- 3D lab : shared helpers ---------- */
function makeRenderer(c){ const r=new THREE.WebGLRenderer({canvas:c,antialias:true,alpha:true}); r.setPixelRatio(Math.min(2,devicePixelRatio||1)); r.outputEncoding=THREE.sRGBEncoding; r.toneMapping=THREE.ACESFilmicToneMapping; r.toneMappingExposure=1.05; return r; }
function fitRenderer(r,c,cam,ortho,viewH){ const b=c.getBoundingClientRect(); if(!b.width) return; const need=Math.round(b.width*r.getPixelRatio())!==c.width||Math.round(b.height*r.getPixelRatio())!==c.height; if(need) r.setSize(b.width,b.height,false); const a=b.width/b.height; if(ortho){ cam.left=-viewH*a/2; cam.right=viewH*a/2; cam.top=viewH/2; cam.bottom=-viewH/2; } else cam.aspect=a; cam.updateProjectionMatrix(); }
function pastelLights(scene){ scene.add(new THREE.HemisphereLight(0xdcd3ff,0xffd7c4,.9)); const key=new THREE.DirectionalLight(0xffffff,.9); key.position.set(4,7,5); scene.add(key); const fill=new THREE.DirectionalLight(0xffb9e3,.35); fill.position.set(-5,3,-4); scene.add(fill); return key; }
function pastelEnv(r){ const es=new THREE.Scene(); const g=new THREE.SphereGeometry(10,32,16); const cols=[]; const pos=g.attributes.position; const top=new THREE.Color(0xcdb9ff), mid=new THREE.Color(0xfff1f6), bot=new THREE.Color(0xffd7c4); for(let i=0;i<pos.count;i++){ const y=pos.getY(i)/10; const c= y>0? mid.clone().lerp(top,y) : mid.clone().lerp(bot,-y); cols.push(c.r,c.g,c.b); } g.setAttribute('color',new THREE.Float32BufferAttribute(cols,3)); es.add(new THREE.Mesh(g,new THREE.MeshBasicMaterial({vertexColors:true,side:THREE.BackSide}))); const panel=(x,y,z,c)=>{ const m=new THREE.Mesh(new THREE.PlaneGeometry(4,2.2),new THREE.MeshBasicMaterial({color:c})); m.position.set(x,y,z); m.lookAt(0,0,0); es.add(m); }; panel(-4,5,3,0xffffff); panel(5,3,-2,0xffe9f3); panel(0,-2,6,0xe6ecff); const p=new THREE.PMREMGenerator(r); p.compileEquirectangularShader(); const t=p.fromScene(es,0.04).texture; p.dispose(); return t; }
function loopCtl(fig,fn){ let running=false, raf=null; const tick=t=>{ fn(t); if(running) raf=requestAnimationFrame(tick); }; const start=()=>{ if(running||REDUCED) return; running=true; raf=requestAnimationFrame(tick); }; const stop=()=>{ running=false; if(raf) cancelAnimationFrame(raf); }; if('IntersectionObserver' in window) new IntersectionObserver(es=>{ es.forEach(e=>{ if(e.isIntersecting) start(); else stop(); }); },{threshold:[0.05]}).observe(fig); return {start,stop,once:()=>fn(performance.now())}; }

function addAurora(fig,stage,seed){ const au=document.createElement('canvas'); au.className='aurora'; stage.insertBefore(au, stage.firstChild); const P=['--p1','--p2','--p3','--p4'].map(cssVar).map(h=>{h=h.replace('#','');return [parseInt(h.slice(0,2),16),parseInt(h.slice(2,4),16),parseInt(h.slice(4,6),16)];}); const ctx=au.getContext('2d'); const LW=96,LH=60; au.width=LW; au.height=LH; let rr=seed*9301+49297; const rnd=()=>{ rr=(rr*9301+49297)%233280; return rr/233280; }; const blobs=[]; for(let i=0;i<5;i++) blobs.push({x:rnd(),y:rnd(),rad:.35+rnd()*.3,sp:.08+rnd()*.12,ph:rnd()*6.28,col:P[(i+seed)%4]}); const t0=performance.now(); const ctl=loopCtl(fig,now=>{ const t=(now-t0)/1000; ctx.fillStyle='#fff'; ctx.fillRect(0,0,LW,LH); blobs.forEach(b=>{ const x=(b.x+Math.sin(t*b.sp+b.ph)*.18)*LW, y=(b.y+Math.cos(t*b.sp*.8+b.ph)*.18)*LH; const g=ctx.createRadialGradient(x,y,0,x,y,b.rad*LW); g.addColorStop(0,'rgba('+b.col.join(',')+',.95)'); g.addColorStop(1,'rgba('+b.col.join(',')+',0)'); ctx.fillStyle=g; ctx.fillRect(0,0,LW,LH); }); }); ctl.once(); ctl.start(); }

/* N · orthographic glass bars */
labs.glass = (function(){ const fig=document.querySelector('[data-tool=glass]'); if(!fig||!window.THREE) return null; const c=fig.querySelector('canvas'), out=fig.querySelector('output');
  const r=makeRenderer(c); r.shadowMap.enabled=true; r.shadowMap.type=THREE.PCFSoftShadowMap;
  const scene=new THREE.Scene(); scene.environment=pastelEnv(r); const cam=new THREE.OrthographicCamera(-1,1,1,-1,.1,100); cam.position.set(7,6,7); cam.lookAt(0,1.1,0);
  const key=pastelLights(scene); key.intensity=1.3; key.castShadow=true; key.shadow.mapSize.set(1024,1024); key.shadow.camera.left=-6; key.shadow.camera.right=6; key.shadow.camera.top=6; key.shadow.camera.bottom=-6; key.shadow.radius=4;
  const ground=new THREE.Mesh(new THREE.PlaneGeometry(30,30), new THREE.ShadowMaterial({opacity:.10})); ground.rotation.x=-Math.PI/2; ground.receiveShadow=true; scene.add(ground);
  const bars=[]; const mat=new THREE.MeshPhysicalMaterial({color:0x3D6BFF, roughness:.28, metalness:0, transmission:.18, clearcoat:1, clearcoatRoughness:.18, transparent:true, opacity:.97, envMapIntensity:1.0});
  DATA.forEach((v,i)=>{ const h=v/MAX*3.4; const g=new THREE.BoxGeometry(.72,1,.72); g.translate(0,.5,0); const m=new THREE.Mesh(g,mat.clone()); m.material.color.setHSL(.64-(i*.014),.9,.58); m.position.set((i-3)*1.05,0,0); m.scale.y=.001; m.userData.h=h; m.castShadow=true; scene.add(m); bars.push(m); });
  const line=new THREE.Line(new THREE.BufferGeometry().setFromPoints(DATA.map((v,i)=>new THREE.Vector3((i-3)*1.05, v/MAX*3.4+.08, .5))), new THREE.LineBasicMaterial({color:0x0B0B0C,transparent:true,opacity:.6})); line.geometry.setDrawRange(0,0); scene.add(line);
  const group=new THREE.Group(); scene.add(group); let tilt=0;
  const ctl=loopCtl(fig,t=>{ fitRenderer(r,c,cam,true,6.4); tilt+=0.0025; cam.position.set(7*Math.cos(tilt*.6),6,7*Math.sin(tilt*.6)+0.001); cam.position.x=7; cam.position.z=7+Math.sin(tilt)*0.6; cam.lookAt(0,1.1,0); r.render(scene,cam); });
  return { play(){ ctl.start(); if(REDUCED){ bars.forEach(b=>b.scale.y=b.userData.h); line.geometry.setDrawRange(0,DATA.length); out.textContent=fmt(KPI); ctl.once(); return; } bars.forEach(b=>b.scale.y=.001); line.geometry.setDrawRange(0,0); tween(900,0,e=>{ bars.forEach((b,i)=>{ const l=Math.min(1,Math.max(0,(e*DATA.length-i)/1.6)); b.scale.y=Math.max(.001,b.userData.h*l); }); line.geometry.setDrawRange(0,Math.round(e*DATA.length)); }); tween(600,400,e=>{ out.textContent=fmt(KPI*e); }); } }; })();

labs.shader = (function(){ const fig=document.querySelector('[data-tool=shader]'); if(!fig||!window.THREE) return null; const c=fig.querySelector('canvas'), out=fig.querySelector('output');
  const r=makeRenderer(c); r.toneMapping=THREE.NoToneMapping; const scene=new THREE.Scene(); const cam=new THREE.OrthographicCamera(-1,1,1,-1,0,1);
  const mat=new THREE.ShaderMaterial({uniforms:{uTime:{value:0},uRes:{value:new THREE.Vector2(1,1)}},
    vertexShader:'void main(){gl_Position=vec4(position,1.0);}',
    fragmentShader:`precision highp float;uniform float uTime;uniform vec2 uRes;
      vec2 h2(vec2 p){p=vec2(dot(p,vec2(127.1,311.7)),dot(p,vec2(269.5,183.3)));return -1.0+2.0*fract(sin(p)*43758.5453);}
      float noise(vec2 p){vec2 i=floor(p),f=fract(p);vec2 u=f*f*(3.0-2.0*f);return mix(mix(dot(h2(i),f),dot(h2(i+vec2(1,0)),f-vec2(1,0)),u.x),mix(dot(h2(i+vec2(0,1)),f-vec2(0,1)),dot(h2(i+vec2(1,1)),f-vec2(1,1)),u.x),u.y);}
      float fbm(vec2 p){float v=0.0,a=0.5;for(int i=0;i<5;i++){v+=a*noise(p);p=p*2.03+vec2(1.7,9.2);a*=0.5;}return v;}
      void main(){vec2 uv=gl_FragCoord.xy/uRes;vec2 p=uv*vec2(uRes.x/uRes.y,1.0);float t=uTime*0.06;
        float n1=fbm(p*1.4+vec2(t,-t*0.7));float n2=fbm(p*2.2-vec2(t*0.5,t));float n3=fbm(p*0.9+n1*0.8+vec2(-t*0.3,t*0.4));
        vec3 lav=vec3(0.80,0.73,1.0),pink=vec3(1.0,0.73,0.89),peach=vec3(1.0,0.84,0.77),sky=vec3(0.75,0.86,1.0),white=vec3(0.99,0.99,0.99);
        vec3 col=mix(sky,lav,smoothstep(0.2,0.8,n1+0.5));col=mix(col,pink,smoothstep(0.35,0.9,n2+0.5));col=mix(col,peach,smoothstep(0.55,1.0,n3+0.55));col=mix(col,white,smoothstep(0.65,1.0,uv.y+n1*0.3)*0.35);
        gl_FragColor=vec4(col,1.0);}`});
  scene.add(new THREE.Mesh(new THREE.PlaneGeometry(2,2),mat));
  // the chart on top, in SVG overlay would be cleaner; here we draw it as a thin 3D line for the test
  const line=new THREE.Line(new THREE.BufferGeometry().setFromPoints(pts().map(q=>new THREE.Vector3((q[0]/W)*2-1,1-(q[1]/H)*2,-0.5))), new THREE.LineBasicMaterial({color:0x0B0B0C,transparent:true,opacity:.75})); line.geometry.setDrawRange(0,0); scene.add(line);
  const t0=performance.now();
  const ctl=loopCtl(fig,t=>{ const b=c.getBoundingClientRect(); if(!b.width) return; if(Math.round(b.width*r.getPixelRatio())!==c.width) r.setSize(b.width,b.height,false); mat.uniforms.uRes.value.set(c.width,c.height); mat.uniforms.uTime.value=(t-t0)/1000; r.render(scene,cam); });
  return { play(){ ctl.start(); if(REDUCED){ line.geometry.setDrawRange(0,7); out.textContent=fmt(KPI); ctl.once(); return; } line.geometry.setDrawRange(0,0); tween(800,0,e=>{ line.geometry.setDrawRange(0,Math.round(e*7)); }); tween(600,400,e=>{ out.textContent=fmt(KPI*e); }); } }; })();

/* P · ambient glass object */
labs.signal = (function(){ const fig=document.querySelector('[data-tool=signal]'); if(!fig||!window.THREE) return null; const c=fig.querySelector('canvas'), out=fig.querySelector('output');
  const stage=fig.querySelector('.stage-area'); stage.classList.add('sky'); addAurora(fig,stage,13);
  const r=makeRenderer(c); const scene=new THREE.Scene(); scene.environment=pastelEnv(r); const cam=new THREE.PerspectiveCamera(30,2,.1,100); cam.position.set(0,0,9);
  pastelLights(scene);
  const geo=new THREE.TorusKnotGeometry(1.5,.48,220,36,2,3);
  const mat=new THREE.MeshPhysicalMaterial({color:0xf4f6ff, roughness:.08, metalness:0, transmission:.55, ior:1.4, clearcoat:1, clearcoatRoughness:.06, transparent:true, opacity:.98, envMapIntensity:1.6});
  const obj=new THREE.Mesh(geo,mat); scene.add(obj);
  let mx=0,my=0; stage.addEventListener('pointermove',e=>{ const b=stage.getBoundingClientRect(); mx=((e.clientX-b.left)/b.width-.5)*2; my=((e.clientY-b.top)/b.height-.5)*2; },{passive:true}); stage.addEventListener('pointerleave',()=>{ mx=0; my=0; });
  let rotY=0;
  const ctl=loopCtl(fig,t=>{ fitRenderer(r,c,cam,false); rotY+=0.004; obj.rotation.y=rotY+mx*.5; obj.rotation.x+= (my*.4-obj.rotation.x)*.06; obj.position.y=Math.sin(t/1200)*.12; r.render(scene,cam); });
  return { play(){ ctl.start(); if(REDUCED){ out.textContent=fmt(KPI); ctl.once(); return; } obj.scale.setScalar(.001); tween(900,0,e=>{ obj.scale.setScalar(e); }); tween(600,400,e=>{ out.textContent=fmt(KPI*e); }); } }; })();
